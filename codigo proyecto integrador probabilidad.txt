tabla_resumen <- data.frame(
  Variable = c(
    "Ventas Totales",
    "Costo Venta Total",
    "Valor Nacional"
  ),
  
  Media = round(c(
    mean(base_300$ventas_totales),
    mean(base_300$costo_venta_total),
    mean(base_300$valor_nacional)
  ),2),
  
  Mediana = round(c(
    median(base_300$ventas_totales),
    median(base_300$costo_venta_total),
    median(base_300$valor_nacional)
  ),2),
  
  Desv_Estandar = round(c(
    sd(base_300$ventas_totales),
    sd(base_300$costo_venta_total),
    sd(base_300$valor_nacional)
  ),2),
  
  Minimo = round(c(
    min(base_300$ventas_totales),
    min(base_300$costo_venta_total),
    min(base_300$valor_nacional)
  ),2),
  
  Maximo = round(c(
    max(base_300$ventas_totales),
    max(base_300$costo_venta_total),
    max(base_300$valor_nacional)
  ),2),
  
  CV = round(c(
    sd(base_300$ventas_totales) /
      mean(base_300$ventas_totales) * 100,
    
    sd(base_300$costo_venta_total) /
      mean(base_300$costo_venta_total) * 100,
    
    sd(base_300$valor_nacional) /
      mean(base_300$valor_nacional) * 100
  ),2)
)

cat("\n====================================\n")
cat("ESTADÍSTICAS DESCRIPTIVAS\n")

print(tabla_resumen)

#########################################################
# TIPO DE MERCADERÍA
#########################################################

freq_tipo <- as.data.frame(
  table(base_300$tipo_merc)
)

names(freq_tipo) <- c(
  "Codigo",
  "Frecuencia"
)

freq_tipo$Tipo <- c(
  "Al por Mayor",
  "Al por Menor"
)

freq_tipo$Porcentaje <- round(
  freq_tipo$Frecuencia /
    sum(freq_tipo$Frecuencia) * 100,
  2
)

cat("\n====================================\n")
cat("TIPO DE MERCADERÍA\n")
cat("====================================\n")

print(freq_tipo)

#########################################################
# BARRAS TIPO MERCADERÍA
#########################################################

ggplot(
  freq_tipo,
  aes(
    x = Tipo,
    y = Frecuencia,
    fill = Tipo
  )
) +
  geom_col(width = 0.6) +
  geom_text(
    aes(
      label = paste0(
        Porcentaje,
        "%"
      )
    ),
    vjust = -0.4,
    size = 5
  ) +
  scale_fill_manual(
    values = c(
      "#2E86C1",
      "#E67E22"
    )
  ) +
  theme_minimal() +
  theme(
    legend.position = "none"
  ) +
  labs(
    title = "Distribución del Tipo de Mercadería",
    x = "",
    y = "Frecuencia"
  )
